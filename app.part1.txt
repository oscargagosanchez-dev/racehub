const cfg=window.RACEHUB_CONFIG||{};
let db=null;
let supabaseActivated=false;

async function activateSupabase(){
 if(supabaseActivated || !cfg.supabaseUrl || !cfg.supabaseAnonKey || !window.supabase) return;
 supabaseActivated=true;
 try{
   db=window.supabase.createClient(cfg.supabaseUrl,cfg.supabaseAnonKey);
   state.backend='supabase';
   state.loading=true;
   render();
   await boot();
 }catch(err){
   console.warn('No se pudo activar Supabase; RaceHub sigue en modo demo.',err);
   db=null;
   state.backend='demo-error';
   useDemo();
   state.loading=false;
   render();
 }
}

function esc(v=''){return String(v).replace(/[&<>"']/g,m=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[m]));}
function saveLocal(){localStorage.setItem('rh-reg',JSON.stringify([...state.registered]));localStorage.setItem('rh-follow',JSON.stringify([...state.following]));}
function dt(iso){const d=new Date(iso);return {day:d.toLocaleDateString('es-ES',{weekday:'short',day:'2-digit',month:'2-digit'}),time:d.toLocaleTimeString('es-ES',{hour:'2-digit',minute:'2-digit'})};}
function goto(v){state.view=v;render();scrollTo({top:0,behavior:'smooth'});}
function toast(msg){const t=document.createElement('div');t.className='toast';t.textContent=msg;document.body.appendChild(t);setTimeout(()=>t.remove(),2400);}
function modal(html){document.querySelector('#modal-root').innerHTML=`<div class="modal-backdrop" onclick="closeModal(event)"><div class="modal">${html}</div></div>`;}
function closeModal(e){if(!e||e.target.classList.contains('modal-backdrop'))document.querySelector('#modal-root').innerHTML='';}

async function boot(){
 if(db){
   const {data:{session}}=await db.auth.getSession(); state.user=session?.user||null;
   db.auth.onAuthStateChange(async(_e,s)=>{state.user=s?.user||null;await loadData();render();});
   await loadData();
 } else {
   useDemo();
 }
 state.loading=false;render();
}
