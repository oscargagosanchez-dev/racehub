function useDemo(){state.events=DEMO_EVENTS;state.communities=DEMO_COMMUNITIES;state.championships=DEMO_CHAMPIONSHIPS;state.streams=DEMO_STREAMS;state.rounds=[];state.registrations=[];state.profiles=[];state.results=[];}
async function loadData(){
 try{
   const [ev,co,ch,ss,rr,rounds,profiles,results,followsAll]=await Promise.all([
     db.from('events').select('*, communities(name)').order('starts_at',{ascending:true}).limit(200),
     db.from('communities').select('*').order('name'),
     db.from('championships').select('*, communities(name)').order('created_at',{ascending:false}).limit(100),
     db.from('stream_status').select('*').eq('is_live',true).limit(50),
     db.from('registrations').select('*').neq('status','cancelled').order('created_at',{ascending:true}).limit(1000),
     db.from('championship_rounds').select('*').order('round_number',{ascending:true}).limit(500),
     db.from('profiles').select('id,nickname,twitch_enabled,twitch_login,youtube_url,discord_url,teamspeak_url').limit(1000),
     db.from('event_results').select('*').order('position',{ascending:true}).limit(2000),
     db.from('follows').select('community_id,user_id').limit(5000)
   ]);
   if(ev.error||co.error||ch.error||rr.error||rounds.error||profiles.error||results.error||followsAll.error) throw ev.error||co.error||ch.error||rr.error||rounds.error||profiles.error||results.error||followsAll.error;
   state.registrations=rr.data||[];
   state.rounds=rounds.data||[];
   state.profiles=profiles.data||[];
   state.results=results.data||[];
   state.followRecords=followsAll.data||[];
   const counts=new Map();
   state.registrations.forEach(r=>counts.set(r.event_id,(counts.get(r.event_id)||0)+1));
   state.events=(ev.data||[]).map(x=>({id:x.id,title:x.title,community:x.communities?.name||'Comunidad',communityId:x.community_id,championshipId:x.championship_id,sim:x.simulator,track:x.track,class:x.car_class||'Multiclase',startsAt:x.starts_at,duration:x.duration_minutes,slots:counts.get(x.id)||0,max:x.max_slots,status:x.status,type:x.event_type,description:x.description||'',createdBy:x.created_by}));
   state.communities=(co.data||[]).map(x=>({id:x.id,name:x.name,sims:x.description||'Simracing',description:x.description||'',country:x.country||'',language:x.language||'es',websiteUrl:x.website_url||'',twitchUrl:x.twitch_url||'',youtubeUrl:x.youtube_url||'',discordUrl:x.discord_url||'',teamspeakUrl:x.teamspeak_url||'',members:state.followRecords.filter(f=>String(f.community_id)===String(x.id)).length,events:state.events.filter(e=>e.communityId===x.id).length,tag:x.name.slice(0,3).toUpperCase(),ownerId:x.owner_id}));
   state.championships=(ch.data||[]).map(x=>({id:x.id,name:x.name,community:x.communities?.name||'Comunidad',communityId:x.community_id,sim:x.simulator,status:x.status,season:x.season||'',description:x.description||'',rounds:state.rounds.filter(r=>r.championship_id===x.id).length,class:x.car_class||'Multiclase',pointsSystem:Array.isArray(x.points_system)?x.points_system:[25,18,15,12,10,8,6,4,2,1]}));
   state.streams=(ss.data||[]).map(x=>{const p=state.profiles.find(p=>p.id===x.user_id);return {name:p?.nickname||'Piloto',game:x.game_name||'Simracing',title:x.title||'En directo',event:state.events.find(e=>e.id===x.event_id)?.title||'Directo',viewers:x.viewer_count||0,url:x.twitch_url||'#'}});
   state.backend='supabase';
   const realEmpty=!state.events.length&&!state.communities.length&&!state.championships.length;
   if(realEmpty){
     state.events=DEMO_EVENTS.map(x=>({...x,isDemo:true}));
     state.communities=DEMO_COMMUNITIES.map(x=>({...x,isDemo:true,ownerId:null}));
     state.championships=DEMO_CHAMPIONSHIPS.map(x=>({...x,isDemo:true,communityId:null,status:'active'}));
     state.streams=DEMO_STREAMS.map(x=>({...x,isDemo:true}));
     state.backend='supabase-empty-demo';
   }
   if(state.user){
     const p=await db.from('profiles').select('*').eq('id',state.user.id).maybeSingle();state.profile=p.data||null;
     state.registered=new Set((rr.data||[]).filter(r=>r.user_id===state.user.id&&r.status!=='cancelled').map(x=>x.event_id));
     state.following=new Set(state.followRecords.filter(f=>f.user_id===state.user.id).map(x=>x.community_id));
   }else{
     state.registered.clear(); state.following.clear(); state.profile=null;
   }
 }catch(err){console.warn('Supabase no disponible, modo demo:',err);state.backend='demo-error';useDemo();}
}

function championshipEvents(champId){return state.events.filter(e=>String(e.championshipId)===String(champId));}
function championshipStandings(champId){
 const evIds=new Set(championshipEvents(champId).map(e=>String(e.id)));
 const rows=state.results.filter(r=>evIds.has(String(r.event_id)));
 const map=new Map();
 rows.forEach(r=>{
   const key=r.user_id||r.driver_name;
   const cur=map.get(key)||{name:r.driver_name,team:r.team_name||'',points:0,wins:0,podiums:0,starts:0};
   cur.points+=Number(r.points||0)+Number(r.bonus_points||0); cur.starts+=1;
   if(Number(r.position)===1)cur.wins+=1; if(Number(r.position)<=3)cur.podiums+=1; map.set(key,cur);
 });
