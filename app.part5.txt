async function createEvent(){
 const c=document.querySelector('#ce-community').value,championship_id=document.querySelector('#ce-championship')?.value||null,title=document.querySelector('#ce-title').value.trim(),sim=document.querySelector('#ce-sim').value.trim(),track=document.querySelector('#ce-track').value.trim(),carClass=document.querySelector('#ce-class').value.trim(),eventType=document.querySelector('#ce-type').value,start=document.querySelector('#ce-start').value,duration=Number(document.querySelector('#ce-duration').value||60),max=Number(document.querySelector('#ce-max').value||40);
 if(!title)return setEventError('Escribe el nombre del evento.');
 if(!track)return setEventError('Escribe el circuito.');
 if(!start)return setEventError('Selecciona la fecha y también la hora del evento.');
 if(!sim)return setEventError('Selecciona o escribe el simulador.');
 if(duration<1)return setEventError('La duración debe ser mayor de 0 minutos.');
 if(max<1)return setEventError('El número de plazas debe ser mayor de 0.');
 const parsedStart=new Date(start);
 if(Number.isNaN(parsedStart.getTime()))return setEventError('La fecha y hora no son válidas.');
 const btn=document.querySelector('#ce-submit'); if(btn){btn.disabled=true;btn.textContent='PUBLICANDO…';}
 try{
   if(db){
     const {error}=await db.from('events').insert({community_id:c,championship_id:championship_id||null,title,simulator:sim,track,car_class:carClass,event_type:eventType,starts_at:parsedStart.toISOString(),duration_minutes:duration,max_slots:max,status:'open',created_by:state.user.id});
     if(error)return setEventError(error.message||'No se ha podido publicar el evento.');
     await loadData();
   }else{
     const community=state.communities.find(x=>String(x.id)===String(c));state.events.unshift({id:'local-'+Date.now(),title,community:community?.name||'Mi comunidad',communityId:c,sim,track,class:carClass,startsAt:parsedStart.toISOString(),duration,slots:0,max,status:'open',type:eventType});
   }
   closeModal();toast('Evento publicado');render();
 }catch(err){setEventError(err?.message||'Error al publicar el evento.');}
 finally{if(btn){btn.disabled=false;btn.textContent='PUBLICAR EVENTO';}}
}
